**Goal:** 用纯 Node 实现单端口协议代理核心：识别协议（Anthropic / OpenAI）→ 查映射表 → 按协议注入认证并改写 model → 转发上游并透传 SSE。curl 即可验证全链路，不依赖 Electron。

**Tech Stack:** Node.js 内置 `http`/`https`/`url`/`crypto`，测试用 `node:test` + `node:assert/strict`。CommonJS，与 CCGUI 主进程风格一致（kebab-case 文件名、单引号、中文注释、文件头职责注释）。

```js
// 协议识别器：根据请求路径、请求头、请求体特征判断请求属于哪种 agent 协议。
// 打分制：anthropic 与 openai 各自累计得分，高分者胜；并列或全零视为无法识别。
// 无法识别时返回 protocol = null，并把各项依据带回去供错误响应与日志使用。

'use strict';

// 路径规则：命中即给该协议加分。分数刻意给得比头/体特征高——路径是最强信号
const PATH_RULES = [
  { protocol: 'anthropic', score: 3, pattern: /^\/v1\/messages(?:\/|$)/ },
  { protocol: 'openai', score: 3, pattern: /^\/v1\/(?:chat\/completions|responses|completions|embeddings|models)(?:\/|$)/ }
];

// 解析请求体为 JSON 对象；失败或非对象返回 null，绝不抛异常
function parseBody(body) {
  if (!body) return null;
  try {
    const parsed = Buffer.isBuffer(body) ? JSON.parse(body.toString('utf8')) : JSON.parse(body);
    return parsed && typeof parsed === 'object' ? parsed : null;
  } catch (err) {
    return null;
  }
}

// 识别入口。入参 headers 需为 Node 原生小写键风格的请求头对象
function detectProtocol({ method, path, headers, body }) {
  const scores = { anthropic: 0, openai: 0 };
  const reasons = [];
  const pathname = String(path || '').split('?')[0];
  const reqHeaders = headers || {};

  for (const rule of PATH_RULES) {
    if (rule.pattern.test(pathname)) {
      scores[rule.protocol] += rule.score;
      reasons.push(`路径 ${pathname} 命中 ${rule.protocol} 路径规则 +${rule.score}`);
    }
  }
  if (reqHeaders['anthropic-version']) {
    scores.anthropic += 3;
    reasons.push('请求头 anthropic-version +3');
  }
  if (reqHeaders['x-api-key']) {
    scores.anthropic += 1;
    reasons.push('请求头 x-api-key +1');
  }
  const payload = parseBody(body);
  if (payload && 'system' in payload && 'max_tokens' in payload) {
    scores.anthropic += 2;
    reasons.push('请求体顶层同时含 system 与 max_tokens +2');
  }

  let protocol = null;
  if (scores.anthropic !== scores.openai) {
    protocol = scores.anthropic > scores.openai ? 'anthropic' : 'openai';
  }
  return { protocol, scores, reasons };
}

module.exports = { detectProtocol, parseBody };
```

```js
// 映射查表：在指定协议的映射表里为客户端模型名找到目标供应商与真实模型名。
// 优先级：精确匹配 > 通配匹配（前缀最长者优先）> 兜底规则（clientModel 为 *）。
// 启停规则：协议停用报 503；供应商或单条映射停用的条目视为不可用直接跳过。

'use strict';

function isWildcardPattern(pattern) {
  return typeof pattern === 'string' && pattern.length > 1 && pattern.endsWith('*');
}

// 单条规则能否匹配客户端模型名（精确 / 通配 / 兜底三种形态）
function matchMapping(mapping, clientModel) {
  const pattern = mapping.clientModel;
  if (typeof pattern !== 'string' || pattern.length === 0) return false;
  if (pattern === '*') return true;
  if (pattern === clientModel) return true;
  if (isWildcardPattern(pattern)) return clientModel.startsWith(pattern.slice(0, -1));
  return false;
}

function routingError(status, code, message) {
  return { error: { status, code, message } };
}

// 查表入口。config 为应用配置对象（结构与 ~/.ccproxy/app.json 一致）
function resolveRouting(config, protocolId, clientModel) {
  const protocol = (config && Array.isArray(config.protocols) ? config.protocols : []).find((p) => p.id === protocolId);
  if (!protocol) {
    return routingError(404, 'protocol_not_found', `配置中不存在协议 ${protocolId}`);
  }
  if (!protocol.enabled) {
    return routingError(503, 'protocol_disabled', `协议 ${protocol.name}（${protocol.id}）已停用`);
  }

  const providers = Array.isArray(protocol.providers) ? protocol.providers : [];
  const mappings = Array.isArray(protocol.mappings) ? protocol.mappings : [];

  // 只保留可用条目：映射启用、供应商存在且启用
  const usable = [];
  for (const mapping of mappings) {
    if (!mapping || !mapping.enabled) continue;
    const provider = providers.find((p) => p.id === mapping.providerId);
    if (!provider || !provider.enabled) continue;
    usable.push({ mapping, provider });
  }

  // 第一轮：精确匹配
  let hit = usable.find((e) => e.mapping.clientModel === clientModel);
  // 第二轮：通配，前缀最长者优先
  if (!hit) {
    const wildcards = usable
      .filter((e) => isWildcardPattern(e.mapping.clientModel) && matchMapping(e.mapping, clientModel))
      .sort((a, b) => (b.mapping.clientModel.length - a.mapping.clientModel.length));
    hit = wildcards[0];
  }
  // 第三轮：兜底
  if (!hit) {
    hit = usable.find((e) => e.mapping.clientModel === '*');
  }

  if (!hit) {
    return routingError(404, 'mapping_not_found', `模型 ${clientModel || '(空)'} 未在 ${protocol.name} 协议映射表中配置`);
  }
  return {
    protocolId,
    provider: { id: hit.provider.id, name: hit.provider.name, baseUrl: hit.provider.baseUrl, token: hit.provider.token },
    mapping: hit.mapping,
    realModel: hit.mapping.realModel
  };
}

module.exports = { resolveRouting, matchMapping };
```

```js
// 协议适配器：收敛「认证怎么注入」「model 字段改在哪」这类协议差异，转发逻辑里不写死任何协议细节。
// 第一版只有直通适配器（上游协议 = 来源协议）。将来做跨协议转换时，
// 在这里新增「目标协议 != 来源协议」的适配器实现即可，路由与转发层不用动。

'use strict';

// 固定剥离清单：无论什么协议，入站请求携带的凭据一律摘掉，保证客户端的假 key 绝不外泄
const STRIP_INBOUND_HEADERS = ['authorization', 'x-api-key', 'api-key', 'x-goog-api-key', 'proxy-authorization'];

// 就地剥离入站凭据（作用于传入对象本身），返回同一对象便于链式调用
function stripInboundCredentials(headers) {
  for (const name of STRIP_INBOUND_HEADERS) {
    delete headers[name];
  }
  return headers;
}

const PROTOCOL_ADAPTERS = {
  anthropic: {
    protocol: 'anthropic',
    // Anthropic 协议规范认证方式：x-api-key 头
    applyAuth(headers, token) {
      headers['x-api-key'] = token;
    },
    rewriteModel(payload, realModel) {
      if (payload && typeof payload === 'object') payload.model = realModel;
      return payload;
    }
  },
  openai: {
    protocol: 'openai',
    // OpenAI 协议规范认证方式：Authorization Bearer
    applyAuth(headers, token) {
      headers.authorization = `Bearer ${token}`;
    },
    rewriteModel(payload, realModel) {
      if (payload && typeof payload === 'object') payload.model = realModel;
      return payload;
    }
  }
};

function getAdapter(protocolId) {
  const adapter = PROTOCOL_ADAPTERS[protocolId];
  if (!adapter) throw new Error(`未知协议: ${protocolId}`);
  return adapter;
}

module.exports = { getAdapter, stripInboundCredentials, STRIP_INBOUND_HEADERS };
```

```js
// 上游转发：拼 URL、发请求、SSE 边收边写、超时三档、错误映射。
// 转发不做任何协议内容解析——响应体原样透传，请求体的改写已在上游调用方完成。

'use strict';

const http = require('http');
const https = require('https');
const { URL } = require('url');

// hop-by-hop 头与每请求必须重算的头，不透传
const HOP_BY_HOP_HEADERS = ['connection', 'keep-alive', 'transfer-encoding', 'upgrade', 'te', 'trailer', 'proxy-authenticate', 'proxy-authorization'];
const REQUEST_HEADER_BLOCKLIST = new Set([...HOP_BY_HOP_HEADERS, 'host', 'content-length', 'accept-encoding']);

// 超时三档：流式首字节 / 流式空闲 / 非流式整体
const FIRST_BYTE_TIMEOUT_MS = 60 * 1000;
const STREAM_IDLE_TIMEOUT_MS = 120 * 1000;
const NON_STREAM_TIMEOUT_MS = 600 * 1000;

// 去掉重复的版本段：baseUrl 已以 v1/v1beta 结尾而剩余路径又带同一段时只保留一份
function dropDuplicatedVersionSegment(baseUrl, remainingPath) {
  const baseSegments = baseUrl.replace(/\/+$/, '').split('/').filter(Boolean);
  const baseTail = baseSegments[baseSegments.length - 1] || '';
  const segments = String(remainingPath || '').split('/').filter(Boolean);
  if (baseTail && segments[0] === baseTail && /^v\d+(?:beta\d*)?$/.test(baseTail)) {
    segments.shift();
  }
  return segments.length ? `/${segments.join('/')}` : '';
}

function buildUpstreamUrl(baseUrl, remainingPath) {
  const base = String(baseUrl || '').replace(/\/+$/, '');
  return `${base}${dropDuplicatedVersionSegment(base, remainingPath)}`;
}

// 出站头清洗：剥 blocklist（键按小写比较；Node 原生请求头本身是小写键）
function sanitizeOutboundHeaders(headers) {
  const outbound = {};
  for (const [name, value] of Object.entries(headers || {})) {
    if (REQUEST_HEADER_BLOCKLIST.has(name.toLowerCase())) continue;
    outbound[name] = value;
  }
  return outbound;
}

// 转发一个请求到上游，把上游响应写回 clientRes。
// options: { url, method, headers, body(Buffer), clientRes, isStreaming, onUpstreamResponse }
// 返回 Promise<{ statusCode }>，错误时已向客户端写出 502 并 resolve（绝不 reject，调用方无需 catch）
function forwardRequest(options) {
  const { url, method, headers, body, clientRes, isStreaming, onUpstreamResponse } = options;
  return new Promise((resolve) => {
    const parsed = new URL(url);
    const transport = parsed.protocol === 'https:' ? https : http;
    let settled = false;
    let responseTimer = null;
    let firstByteTimer = null;

    // 统一清理两档计时器；无论成功还是失败都必须清，否则失败路径会泄漏 60s 计时器
    const clearTimers = () => {
      if (responseTimer) clearTimeout(responseTimer);
      if (firstByteTimer) clearTimeout(firstByteTimer);
    };
    const finish = (result) => {
      if (settled) return;
      settled = true;
      clearTimers();
      resolve(result);
    };
    const fail = (err) => {
      if (settled) return;
      if (!clientRes.headersSent) {
        clientRes.writeHead(502, { 'content-type': 'application/json' });
        clientRes.end(JSON.stringify({ error: { type: 'ccproxy_upstream', message: `上游请求失败: ${err.message}` } }));
      } else {
        clientRes.destroy();
      }
      finish({ statusCode: 502, error: err.message });
    };

    const upstreamReq = transport.request(
      {
        hostname: parsed.hostname,
        port: parsed.port || (parsed.protocol === 'https:' ? 443 : 80),
        path: `${parsed.pathname}${parsed.search}`,
        method,
        headers
      },
      (upstreamRes) => {
        clearTimeout(firstByteTimer);
        if (onUpstreamResponse) onUpstreamResponse(upstreamRes);
        clientRes.writeHead(upstreamRes.statusCode, upstreamRes.headers);

        if (isStreaming) {
          // 流式：空闲计时，每收到一块数据就重置
          const resetIdle = () => {
            if (responseTimer) clearTimeout(responseTimer);
            responseTimer = setTimeout(() => {
              upstreamReq.destroy(new Error(`流式响应空闲超过 ${STREAM_IDLE_TIMEOUT_MS}ms`));
            }, STREAM_IDLE_TIMEOUT_MS);
          };
          resetIdle();
          upstreamRes.on('data', resetIdle);
        } else {
          // 非流式：整体超时
          responseTimer = setTimeout(() => {
            upstreamReq.destroy(new Error(`非流式响应超过 ${NON_STREAM_TIMEOUT_MS}ms`));
          }, NON_STREAM_TIMEOUT_MS);
        }

        upstreamRes.on('data', (chunk) => clientRes.write(chunk));
        upstreamRes.on('end', () => {
          clientRes.end();
          finish({ statusCode: upstreamRes.statusCode });
        });
        upstreamRes.on('error', (err) => {
          clientRes.destroy();
          finish({ statusCode: upstreamRes.statusCode, error: err.message });
        });
      }
    );

    firstByteTimer = setTimeout(() => {
      upstreamReq.destroy(new Error(`上游 ${FIRST_BYTE_TIMEOUT_MS}ms 内无响应`));
    }, FIRST_BYTE_TIMEOUT_MS);

    upstreamReq.on('error', fail);
    if (body) upstreamReq.write(body);
    upstreamReq.end();
  });
}

module.exports = { buildUpstreamUrl, sanitizeOutboundHeaders, forwardRequest, REQUEST_HEADER_BLOCKLIST };
```

```js
// 本地代理服务：单端口接收所有协议请求，识别协议 → 查映射表 → 协议适配改写 → 转发上游。
// 配置通过注入的 getConfig() 每请求现读，改配置即时生效，无需重启代理。

'use strict';

const http = require('http');
const { detectProtocol, parseBody } = require('./detector');
const { resolveRouting } = require('./router');
const { getAdapter, stripInboundCredentials } = require('./adapters');
const { buildUpstreamUrl, sanitizeOutboundHeaders, forwardRequest } = require('./upstream');

const MAX_BODY_BYTES = 64 * 1024 * 1024;

// 读完整请求体；超过上限报 413
function readRequestBody(req) {
  return new Promise((resolve, reject) => {
    const chunks = [];
    let size = 0;
    req.on('data', (chunk) => {
      size += chunk.length;
      if (size > MAX_BODY_BYTES) {
        reject(Object.assign(new Error('请求体超过 64MB 上限'), { statusCode: 413 }));
        req.destroy();
        return;
      }
      chunks.push(chunk);
    });
    req.on('end', () => resolve(Buffer.concat(chunks)));
    req.on('error', reject);
  });
}

// 统一的 JSON 错误响应
function sendJsonError(res, statusCode, payload) {
  if (res.headersSent) {
    res.destroy();
    return;
  }
  res.writeHead(statusCode, { 'content-type': 'application/json' });
  res.end(JSON.stringify(payload));
}

// 单个请求的完整处理链
async function handleRequest(req, res, options) {
  const { getConfig, onLog } = options;
  const startedAt = Date.now();
  const body = await readRequestBody(req);

  const detection = detectProtocol({ method: req.method, path: req.url, headers: req.headers, body });
  if (!detection.protocol) {
    sendJsonError(res, 400, {
      error: { type: 'ccproxy_detect', message: '无法识别请求所属协议', detail: { scores: detection.scores, reasons: detection.reasons } }
    });
    if (onLog) onLog({ timestamp: startedAt, protocol: null, clientModel: null, statusCode: 400, durationMs: Date.now() - startedAt, detection });
    return;
  }

  const payload = parseBody(body);
  const clientModel = payload && typeof payload.model === 'string' ? payload.model : '';
  const config = getConfig();
  const routing = resolveRouting(config, detection.protocol, clientModel);
  if (routing.error) {
    sendJsonError(res, routing.error.status, { error: { type: routing.error.code, message: routing.error.message } });
    if (onLog) onLog({ timestamp: startedAt, protocol: detection.protocol, clientModel, statusCode: routing.error.status, durationMs: Date.now() - startedAt, detection });
    return;
  }

  const adapter = getAdapter(detection.protocol);
  const provider = routing.provider;

  // 改写 model 字段；只有真的变了才重新序列化，没变就原样字节透传
  let forwardBody = body;
  if (payload && payload.model !== routing.realModel) {
    adapter.rewriteModel(payload, routing.realModel);
    forwardBody = Buffer.from(JSON.stringify(payload), 'utf8');
  }

  // 出站头：清洗 → 剥离入站凭据 → 注入供应商真实凭据 → 重算 content-length
  const outboundHeaders = stripInboundCredentials(sanitizeOutboundHeaders(req.headers));
  adapter.applyAuth(outboundHeaders, provider.token);
  outboundHeaders['content-length'] = String(forwardBody.length);

  const isStreaming = !!(payload && payload.stream === true) || String(req.headers.accept || '').includes('text/event-stream');
  const url = buildUpstreamUrl(provider.baseUrl, req.url);

  const result = await forwardRequest({
    url,
    method: req.method,
    headers: outboundHeaders,
    body: forwardBody,
    clientRes: res,
    isStreaming,
    onUpstreamResponse: () => {}
  });

  if (onLog) {
    onLog({
      timestamp: startedAt,
      protocol: detection.protocol,
      clientModel,
      mappingId: routing.mapping.id,
      providerName: provider.name,
      realModel: routing.realModel,
      statusCode: result.statusCode,
      durationMs: Date.now() - startedAt,
      detection
    });
  }
}

function createProxyServer(options = {}) {
  const handleOptions = { getConfig: options.getConfig, onLog: options.onLog };
  return http.createServer((req, res) => {
    handleRequest(req, res, handleOptions).catch((err) => {
      sendJsonError(res, err.statusCode || 500, { error: { type: 'ccproxy_internal', message: err.message } });
    });
  });
}

module.exports = { createProxyServer, readRequestBody, sendJsonError };
```

```js
// 代理生命周期：启动、停止、状态查询。
// 监听固定 127.0.0.1；端口占用时直接报错，不静默顺延（界面据此提示改端口）。

'use strict';

const { createProxyServer } = require('./server');

function createLocalProxy({ getConfig, onLog }) {
  let server = null;

  return {
    start() {
      return new Promise((resolve, reject) => {
        if (server) {
          resolve({ port: server.address().port });
          return;
        }
        const srv = createProxyServer({ getConfig, onLog });
        const port = getConfig().settings.port;
        srv.once('error', (err) => {
          server = null;
          reject(err);
        });
        srv.listen(port, '127.0.0.1', () => {
          server = srv;
          resolve({ port: srv.address().port });
        });
      });
    },

    stop() {
      return new Promise((resolve) => {
        if (!server) {
          resolve();
          return;
        }
        server.close(() => {
          server = null;
          resolve();
        });
      });
    },

    getStatus() {
      return {
        running: !!server,
        port: server ? server.address().port : (getConfig().settings.port || null)
      };
    }
  };
}
```
